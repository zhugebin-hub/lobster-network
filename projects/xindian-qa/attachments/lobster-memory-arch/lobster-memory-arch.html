4.3 教材入库流程

# 教材入库脚本 # 文件: scripts/ingest-textbooks.sh #!/bin/bash # 10本教材 × 10万字 → 百炼知识库 TEXTBOOKS=( "教材01_基础理论.pdf" "教材02_核心技术.pdf" "教材03_实践应用.pdf" # ... 共10本 ) for book in "${TEXTBOOKS[@]}"; do echo "📖 处理: $book" # 步骤1: PDF 解析 → Markdown python3 scripts/pdf_to_md.py "$book" # 步骤2: 智能分块 (800字/块, 200字重叠) python3 scripts/chunk_text.py "${book%.pdf}.md" \ --chunk-size 800 --overlap 200 # 步骤3: 上传百炼知识库 curl -X POST "https://bailian.aliyuncs.com/api/v1/knowledge/documents" \ -H "Authorization: Bearer $BAILIAN_API_KEY" \ -F "file=@chunks/${book%.pdf}_chunks.jsonl" \ -F "kb_id=kb-xxxxxxxx" echo "✅ $book 入库完成" done
